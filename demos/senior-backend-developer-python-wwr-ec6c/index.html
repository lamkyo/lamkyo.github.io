Optimized Django REST Framework ViewSet with atomic transactions, Redis caching, and robust error handling.

api/views.py Python 3.11 • Django 4.2 LTS 
from django.db import transaction
from django.core.cache import cache
from rest_framework import viewsets, status
from rest_framework.response import Response
from rest_framework.permissions import IsAuthenticated
from .models import ProjectDeployment
from .serializers import DeploymentSerializer
import logging

logger = logging.getLogger(__name__)

class DeploymentViewSet(viewsets.ModelViewSet):
    """
    High-performance ViewSet for managing client cloud deployments.
    Features Redis response caching, atomic database operations, and structured logging.
    """
    serializer_class = DeploymentSerializer
    permission_classes = [IsAuthenticated]
    lookup_field = 'uuid'

    def get_queryset(self):
        user = self.request.user
        cache_key = f"deployments_user_{user.id}"
        
        # Try fetching from Redis cache (TTL: 300s)
        queryset = cache.get(cache_key)
        if queryset is None:
            queryset = ProjectDeployment.objects.select_related('client').filter(
                client=user, is_active=True
            ).order_by('-created_at')
            cache.set(cache_key, queryset, timeout=300)
            
        return queryset

    @transaction.atomic
    def create(self, request, *args, **kwargs):
        serializer = self.get_serializer(data=request.data)
        serializer.is_valid(raise_exception=True)
        
        try:
            deployment = serializer.save(created_by=request.user)
            
            # Invalidate user cache upon state mutation
            cache.delete(f"deployments_user_{request.user.id}")
            
            logger.info(f"Successfully deployed project {deployment.uuid} for user {request.user.email}")
            return Response(serializer.data, status=status.HTTP_201_CREATED)
            
        except Exception as e:
            logger.error(f"Deployment failed: {str(e)}", exc_info=True)
            return Response({"error": "Internal deployment error occurred."}, status=status.HTTP_500_INTERNAL_SERVER_ERROR)
