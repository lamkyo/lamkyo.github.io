+"""Optional MCP tool-execution profile: maps host-verified MCP tool operations onto the
+protocol-neutral ReplicaAuthorizer seam in ports.py.
+
+Pinned specification revision: MCP 2025-06-18 (tools/list, tools/call). No MCP SDK is
+imported here; the host owns the MCP server, its transport (stdio or Streamable HTTP),
+and its user IAM. LETS stays protocol-neutral and only sees the authorization calls
+below, so this module adds no core runtime dependency.
+
+Boundaries this profile deliberately does not cross:
+
+- discovery (tools/list) grants no authority; a tool is usable only after the host
+  intersects the advertised surface with host policy, the LETS lease capability/residual,
+  the receipt audience policy, and executor trust;
+- the host performs permission/confirmation gates before calling authorize();
+- the executor calls verify_and_claim() immediately before the tool effect;
+- no credentials, bearer tokens, or opaque agent state are passed through the adapter.
+"""
+
+from __future__ import annotations
+
+import hashlib
+import json
+from collections.abc import Mapping
+from dataclasses import dataclass
+from typing import Any
+
+from lets.errors import ConflictError, PolicyError, ValidationError
+from lets.ids import require_digest, require_identifier
+from lets.integrations.ports import ReplicaAuthorizer
+
+WireObject = Mapping[str, Any]
+
+MCP_SPEC_REVISION = "2025-06-18"
+
+
+def effect_digest(tool: str, scope: str, arguments: Mapping[str, Any]) -> str:
+    """Stable digest binding a tool request to its exact arguments."""
+    require_identifier(tool, field="tool")
+    require_identifier(scope, field="scope")
+    if not isinstance(arguments, Mapping):
+        raise ValidationError("tool arguments must be a mapping")
+    encoded = json.dumps(arguments, sort_keys=True, separators=(",", ":"), default=str)
+    return require_digest(
+        "sha256:" + hashlib.sha256(encoded.encode("utf-8")).hexdigest(),
+        field="effect digest",
+    )
+
+
+@dataclass(frozen=True, slots=True)
+class MCPToolBinding:
+    """Host-declared mapping from one MCP tool to its LETS transition and audience."""
+
+    tool: str
+    scope: str
+    transition: str
+    executor_audience: str
+
+    def __post_init__(self) -> None:
+        for field, value in (
+            ("tool", self.tool),
+            ("scope", self.scope),
+            ("transition", self.transition),
+            ("executor_audience", self.executor_audience),
+        ):
+            require_identifier(value, field=f"MCP tool binding {field}")
+
+
+class MCPToolProfile:
+    """Thin adapter: MCP tool call -> LETS transition receipt.
+
+    The host keeps the MCP server, the user IAM, and the tool effect. This class only
+    turns a host-verified tool call into an authorize_effect() request with a stable
+    request ID and an argument digest, and fails closed on retry drift.
+    """
+
+    def __init__(
+        self,
+        authorizer: ReplicaAuthorizer,
+        bindings: tuple[MCPToolBinding, ...] | list[MCPToolBinding],
+    ) -> None:
+        if not bindings:
+            raise ValidationError("MCP tool profile requires at least one tool binding")
+        self.authorizer = authorizer
+        self.bindings = {binding.tool: binding for binding in bindings}
+        if len(self.bindings) != len(bindings):
+            raise ValidationError("MCP tool profile bindings must be unique per tool")
+        self._requests: dict[str, str] = {}
+
+    def discover(self) -> tuple[WireObject, ...]:
+        """Advertised tool surface. Discovery grants no authority."""
+        return tuple(
+            {
+                "tool": binding.tool,
+                "scope": binding.scope,
+                "spec_revision": MCP_SPEC_REVISION,
+            }
+            for binding in sorted(self.bindings.values(), key=lambda item: item.tool)
+        )
+
+    def _binding(self, tool: str, scope: str) -> MCPToolBinding:
+        require_identifier(tool, field="tool")
+        require_identifier(scope, field="scope")
+        binding = self.bindings.get(tool)
+        if binding is None:
+            raise PolicyError(f"unknown MCP tool: {tool}")
+        if binding.scope != scope:
+            raise PolicyError(f"scope {scope} is not granted for MCP tool {tool}")
+        return binding
+
+    def _bind_request(self, request_id: str, digest: str) -> None:
+        require_identifier(request_id, field="request_id")
+        previous = self._requests.get(request_id)
+        if previous is None:
+            self._requests[request_id] = digest
+        elif previous != digest:
+            raise ConflictError(
+                f"request_id {request_id} was already used with different arguments"
+            )
+
+    def authorize_tool_call(
+        self,
+        *,
+        request_id: str,
+        lease_id: str,
+        tool: str,
+        scope: str,
+        arguments: Mapping[str, Any],
+        nonce: str,
+        tenant_id: str,
+        executor_audience: str,
+        expected_sequence: int | None = None,
+    ) -> WireObject:
+        """Authorize one protected tool effect.
+
+        Call this immediately before the effect, then have the executor
+        verify_and_claim() the returned receipt in the effect's own transaction.
+        """
+        binding = self._binding(tool, scope)
+        if tenant_id != self.authorizer.profile.tenant_id:
+            raise PolicyError("tool call tenant does not match the configured profile")
+        if executor_audience != binding.executor_audience:
+            raise PolicyError("tool call audience does not match the tool binding")
+        digest = effect_digest(tool, scope, arguments)
+        self._bind_request(request_id, digest)
+        return self.authorizer.authorize_effect(
+            request_id=request_id,
+            lease_id=lease_id,
+            transition=binding.transition,
+            executor_audience=binding.executor_audience,
+            nonce=nonce,
+            evidence={"tool": tool, "scope": scope, "effect_digest": digest},
+            expected_sequence=expected_sequence,
+        )
+
+    def cancel(self, *, request_id: str, lease_id: str, reason: str) -> WireObject:
+        """Cancel a pending tool call by revoking its lease branch."""
+        return self.authorizer.revoke(lease_id, request_id=request_id, reason=reason)
