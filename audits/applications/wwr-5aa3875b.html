## 3. Production-Ready Code / Architectural Proof

Idempotent, HMAC-verified Shopify webhook handler (TypeScript / Remix) — the backbone of any reliable Shopify integration:

```typescript
// app/routes/webhooks.shopify.ts
import { createHmac, timingSafeEqual } from "node:crypto";
import type { ActionFunctionArgs } from "@remix-run/node";

const SHOPIFY_WEBHOOK_SECRET = process.env.SHOPIFY_WEBHOOK_SECRET!;

/**
 * Verify Shopify's HMAC-SHA256 signature.
 * Uses timingSafeEqual to prevent timing attacks; both buffers
 * must be equal length or timingSafeEqual throws.
 */
function verifyWebhook(rawBody: string, hmacHeader: string): boolean {
  const digest = createHmac("sha256", SHOPIFY_WEBHOOK_SECRET)
    .update(rawBody, "utf8")
    .digest("base64");
  const a = Buffer.from(digest);
  const b = Buffer.from(hmacHeader);
  return a.length === b.length && timingSafeEqual(a, b);
}

export async function action({ request }: ActionFunctionArgs) {
  const rawBody = await request.text(); // raw text, NOT parsed JSON — signature covers exact bytes

  const hmac = request.headers.get("x-shopify-hmac-sha256") ?? "";
  if (!verifyWebhook(rawBody, hmac)) {
    console.warn("webhook_rejected: invalid hmac");
    return new Response("Unauthorized", { status: 401 });
  }

  // Shopify delivers at-least-once: dedupe on event id.
  const topic = request.headers.get("x-shopify-topic") ?? "";
  const eventId = request.headers.get("x-shopify-webhook-id") ?? crypto.randomUUID();
  const payload = JSON.parse(rawBody);

  // Redis SETNX with TTL: returns true only for the first consumer of this event.
  const isNew = await redis.set(`wh:${eventId}`, 1, { nx: true, ex: 86400 });
  if (!isNew) {
    // Duplicate delivery — acknowledge so Shopify stops retrying.
    return new Response(null, { status: 200 });
  }

  try {
    switch (topic) {
      case "products/update":
        // Tag-based edge cache purge so the storefront reflects changes immediately.
        await purgeCacheTags([`product:${payload.id}`, "collection-index"]);
        break;
      case "inventory_levels/update":
        // Edge case: variant may be out of stock — invalidate PDP + cart drawer.
        await purgeCacheTags([`product:${payload.inventory_item_id}`, "cart-hints"]);
        break;
      case "app/uninstalled":
        // Critical edge case: shop revoked access — stop all background jobs.
        await revokeShopTokens(payload.shop_domain);
        break;
      default:
        console.info(`unhandled_topic: ${topic}`);
    }
    return new Response(null, { status: 200 });
  } catch (err) {
    // Release the dedupe lock so Shopify's retry can be processed after the failure.
    await redis.del(`wh:${eventId}`);
    console.error("webhook_processing_error", { topic, eventId, err });
    // 500 triggers Shopify's exponential-backoff retry (19 retries over 48h).
    return new Response("Internal Error", { status: 500 });
  }
}
```

Key edge cases handled: raw-byte signature verification, timing-attack resistance, at-least-once deduplication, lock release on failure so retries succeed, and uninstall revocation.
