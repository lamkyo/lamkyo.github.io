## 3. Production-Ready Code / Architectural Proof

Below is an idiomatic Go backend service implementation demonstrating a high-performance, concurrent capacity planning allocation check using Postgres with context timeouts, proper error handling, and structured logging.

```go
package main

import (
	"context"
	"errors"
	"fmt"
	"log/slog"
	"net/http"
	"os"
	"time"

	"github.com/jackc/pgx/v5/pgxpool"
)

// CapacityService handles resource allocation and schedule validation for Toggl 2.0.
type CapacityService struct {
	db     *pgxpool.Pool
	logger *slog.Logger
}

type AllocationRequest struct {
	UserID    string    `json:"user_id"`
	ProjectID string    `json:"project_id"`
	StartTime time.Time `json:"start_time"`
	EndTime   time.Time `json:"end_time"`
	Capacity  int       `json:"capacity_hours"`
}

var ErrOverCapacity = errors.New("resource exceeds maximum capacity for the given timeframe")

func NewCapacityService(db *pgxpool.Pool, logger *slog.Logger) *CapacityService {
	return &CapacityService{db: db, logger: logger}
}

// ValidateAndAllocate checks existing allocations, prevents double-booking/over-capacity,
// and persists the new allocation atomically.
func (s *CapacityService) ValidateAndAllocate(ctx context.Context, req AllocationRequest) error {
	ctx, cancel := context.WithTimeout(ctx, 3*time.Second)
	defer cancel()

	tx, err := s.db.Begin(ctx)
	if err != nil {
		return fmt.Errorf("failed to begin transaction: %w", err)
	}
	defer tx.Rollback(ctx)

	// Query existing allocations within the timeframe using a locking read
	var totalAllocated int
	query := `
		SELECT COALESCE(SUM(capacity_hours), 0)
		FROM resource_allocations
		للWHERE user_id = $1 
		  AND start_time < $3 
		  AND end_time > $2
		FOR UPDATE;
	`

	err = tx.QueryRow(ctx, query, req.UserID, req.StartTime, req.EndTime).Scan(&totalAllocated)
	if err != nil {
		return fmt.Errorf("failed to query existing capacity: %w", err)
	}

	const maxWeeklyCapacity = 40
	if totalAllocated+req.Capacity > maxWeeklyCapacity {
		s.logger.Warn("capacity limit breached", 
			slog.String("user_id", req.UserID),
			slog.Int("current_allocated", totalAllocated),
			slog.Int("requested", req.Capacity),
		)
		return ErrOverCapacity
	}

	insertQuery := `
		INSERT INTO resource_allocations (user_id, project_id, start_time, end_time, capacity_hours)
		VALUES ($1, $2, $3, $4, $5);
	`
	_, err = tx.Exec(ctx, insertQuery, req.UserID, req.ProjectID, req.StartTime, req.EndTime, req.Capacity)
	if err != nil {
		return fmt.Errorf("failed to insert allocation: %w", err)
	}

	if err := tx.Commit(ctx); err != nil {
		return fmt.Errorf("failed to commit transaction: %w", err)
	}

	s.logger.Info("successfully allocated capacity", slog.String("user_id", req.UserID), slog.String("project_id", req.ProjectID))
	return nil
}

func main() {
	logger := slog.New(slog.NewJSONHandler(os.Stdout, nil))
	logger.Info("Initializing Toggl 2.0 Capacity Service Mock")
}
```
