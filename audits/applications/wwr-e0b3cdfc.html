## 3. Production-Ready Code / Architectural Proof

Below is an idiomatic, production-grade Go implementation of a resilient, stateful Agent Execution Loop. It incorporates structured logging, tool-calling dispatch, context propagation, and error handling designed to run reliably within Sticker Mule’s backend infrastructure.

```go
package agent

import (
	"context"
	"encoding/json"
	"fmt"
	"log/slog"
	"time"

	"github.com/google/uuid"
)

// Tool defines the interface that all agent-accessible tools must implement.
type Tool interface {
	Name() string
	Description() string
	Execute(ctx context.Context, input json.RawMessage) (any, error)
}

// AgentTask represents an autonomous unit of work dispatched to the agent.
type AgentTask struct {
	ID           uuid.UUID
	Department   string
	Prompt       string
	MaxIterations int
}

// LLMClient abstracts the underlying foundation model (Claude, OpenAI, Grok).
type LLMClient interface {
	Complete(ctx context.Context, prompt string, availableTools []Tool) (string, []ToolCall, error)
}

type ToolCall struct {
	ToolName string          `json:"tool_name"`
	Input    json.RawMessage `json:"input"`
}

// AutonomousAgentRuntime manages the execution lifecycle of an AI agent.
type AutonomousAgentRuntime struct {
	llmClient     LLMClient
	toolRegistry  map[string]Tool
	logger        *slog.Logger
	maxIterations int
}

func NewAutonomousAgentRuntime(client LLMClient, logger *slog.Logger) *AutonomousAgentRuntime {
	return &AutonomousAgentRuntime{
		llmClient:    client,
		toolRegistry: make(map[string]Tool),
		logger:       logger,
	}
}

func (r *AutonomousAgentRuntime) RegisterTool(tool Tool) {
	r.toolRegistry[tool.Name()] = tool
	r.logger.Info("Registered tool for agent use", "tool", tool.Name())
}

// ExecuteLoop runs the agent autonomously until a stopping condition is met or max iterations are reached.
func (r *AutonomousAgentRuntime) ExecuteLoop(ctx context.Context, task AgentTask) (string, error) {
	r.logger.Info("Starting autonomous agent execution", "task_id", task.ID, "department", task.Department)

	currentPrompt := task.Prompt
	var tools []Tool
	for _, t := range r.toolRegistry {
		tools = append(tools, t)
	}

	for iteration := 1; iteration <= task.MaxIterations; iteration++ {
		select {
		case <-ctx.Done():
			return "", fmt.Errorf("agent execution canceled: %w", ctx.Err())
		default:
		}

		r.logger.Debug("Executing agent reasoning step", "iteration", iteration)

		response, toolCalls, err := r.llmClient.Complete(ctx, currentPrompt, tools)
		if err != nil {
			r.logger.Error("LLM completion failed", "error", err, "iteration", iteration)
			return "", fmt.Errorf("llm completion error: %w", err)
		}

		// If no tools were invoked, the agent has reached its final answer.
		if len(toolCalls) == 0 {
			r.logger.Info("Agent completed task successfully", "iterations_used", iteration)
			return response, nil
		}

		// Execute requested tools and feed results back into the context.
		for _, call := range toolCalls {
			tool, exists := r.toolRegistry[call.ToolName]
			if !exists {
				currentPrompt += fmt.Sprintf("\nSystem Error: Tool '%s' does not exist.", call.ToolName)
				continue
			}

			r.logger.Info("Invoking tool", "tool", call.ToolName)
			start := time.Now()
			result, err := tool.Execute(ctx, call.Input)
			duration := time.Since(start)

			if err != nil {
				r.logger.Error("Tool execution failed", "tool", call.ToolName, "error", err, "duration_ms", duration.Milliseconds())
				currentPrompt += fmt.Sprintf("\nTool %s execution failed: %v", call.ToolName, err)
				continue
			}

			resultBytes, _ := json.Marshal(result)
			currentPrompt += fmt.Sprintf("\nTool %s output: %s", call.ToolName, string(resultBytes))
		}
	}

	return "", fmt.Errorf("agent exceeded maximum iterations (%d) without completing task", task.MaxIterations)
}
```
