## 3. Architectural Proof: Idempotent Incident State Machine

```python
"""
Idempotent incident state transition handler for Datadog webhooks.

Edge cases handled:
- Duplicate webhook deliveries (at-least-once semantics)
- Out-of-order events (stale revision rejection)
- Partial API failures (safe retry with backoff)
"""
import time
from dataclasses import dataclass
from enum import Enum
from typing import Optional

import requests

VALID_TRANSITIONS = {
    "triggered": {"acknowledged", "resolved"},
    "acknowledged": {"resolving", "resolved"},
    "resolving": {"resolved"},
    "resolved": set(),  # terminal state
}


@dataclass(frozen=True)
class IncidentEvent:
    incident_id: str
    revision: int          # monotonic version for optimistic concurrency
    target_state: str
    dedup_key: str         # idempotency key from webhook delivery
    payload: dict


class TransitionResult(Enum):
    APPLIED = "applied"
    DUPLICATE = "duplicate"      # already processed this dedup key
    STALE = "stale"              # event older than current revision
    INVALID = "invalid"          # illegal state transition


class IncidentStateMachine:
    def __init__(self, datadog_api: str, api_key: str, app_key: str):
        self._session = requests.Session()
        self._session.headers.update({
            "DD-API-KEY": api_key,
            "DD-APPLICATION-KEY": app_key,
        })
        self._api = datadog_api
        self._processed_keys: set[str] = set()  # prod: Redis with TTL

    def handle(self, event: IncidentEvent) -> TransitionResult:
        # 1. Idempotency: at-least-once delivery means duplicates are expected.
        if event.dedup_key in self._processed_keys:
            return TransitionResult.DUPLICATE

        # 2. Fetch current state; 404 means incident not yet created upstream.
        resp = self._session.get(
            f"{self._api}/api/v2/incidents/{event.incident_id}", timeout=10
        )
        if resp.status_code == 404:
            return TransitionResult.INVALID
        resp.raise_for_status()
        current = resp.json()["data"]["attributes"]

        # 3. Optimistic concurrency: reject out-of-order events rather than
        #    clobbering newer state (e.g., a delayed 'acknowledge' arriving
        #    after 'resolve').
        if event.revision < current.get("revision", 0):
            return TransitionResult.STALE

        # 4. Validate the transition against the state machine.
        if event.target_state not in VALID_TRANSITIONS.get(current["state"], set()):
            return TransitionResult.INVALID

        # 5. Apply with retry/backoff; safe because the operation is idempotent.
        for attempt in range(5):
            try:
                patch = self._session.patch(
                    f"{self._api}/api/v2/incidents/{event.incident_id}",
                    json={"data": {"attributes": {
                        "state": event.target_state,
                        "revision": event.revision,
                    }}},
                    timeout=10,
                )
                if patch.status_code == 409:  # lost the CAS race
                    return TransitionResult.STALE
                patch.raise_for_status()
                self._processed_keys.add(event.dedup_key)
                return TransitionResult.APPLIED
            except requests.RequestException:
                if attempt == 4:
                    raise  # surface to DLQ in production
                time.sleep(2 ** attempt)  # exponential backoff
        return TransitionResult.INVALID  # unreachable, satisfies type checker
```
