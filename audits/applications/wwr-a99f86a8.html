## 3. Production-Ready Code / Architectural Proof

Below is an idiomatic, production-ready Go implementation of a robust Webhook Dispatcher and Idempotent Retry Engine—representative of the robust integration machinery required when communicating with banks and third-party software houses.

```go
package main

import (
	"bytes"
	"context"
	"crypto/hmac"
	"crypto/sha256"
	"encoding/hex"
	"errors"
	"fmt"
	"log"
	"net/http"
	"sync"
	"time"
)

// WebhookEvent represents an outbound notification to a merchant or bank.
type WebhookEvent struct {
	ID        string    `json:"id"`
	EventType string    `json:"event_type"`
	Payload   string    `json:"payload"`
	Timestamp time.Time `json:"timestamp"`
}

// Dispatcher manages concurrent, retrying webhook deliveries with exponential backoff.
type Dispatcher struct {
	client *http.Client
	secret string
}

func NewDispatcher(secret string, timeout time.Duration) *Dispatcher {
	return &Dispatcher{
		client: &http.Client{Timeout: timeout},
		secret: secret,
	}
}

// ComputeSignature generates an HMAC-SHA256 signature for payload verification.
func (d *Dispatcher) ComputeSignature(payload []byte) string {
	mac := hmac.New(sha256.New, []byte(d.secret))
	mac.Write(payload)
	return hex.EncodeToString(mac.Sum(nil))
}

// SendWithRetry delivers a webhook with jittered exponential backoff.
func (d *Dispatcher) SendWithRetry(ctx context.Context, targetURL string, event WebhookEvent, maxRetries int) error {
	body := []byte(event.Payload)
	signature := d.ComputeSignature(body)

	backoff := 500 * time.Millisecond

	for attempt := 1; attempt <= maxRetries; attempt++ {
		select {
		case <-ctx.Done():
			return ctx.Err()
		default:
		}

		req, err := http.NewRequestWithContext(ctx, http.MethodPost, targetURL, bytes.NewBuffer(body))
		if err != nil {
			return fmt.Errorf("failed to create request: %w", err)
		}

		req.Header.Set("Content-Type", "application/json")
		req.Header.Set("X-Twikey-Event-ID", event.ID)
		req.Header.Set("X-Twikey-Signature", signature)

		resp, err := d.client.Do(req)
		if err == nil && resp.StatusCode >= 200 && resp.StatusCode < 300 {
			resp.Body.Close()
			log.Printf("[INFO] Webhook %s delivered successfully to %s on attempt %d", event.ID, targetURL, attempt)
			return nil
		}

		if resp != nil {
			resp.Body.Close()
			log.Printf("[WARN] Webhook %s delivery failed with status %d on attempt %d", event.ID, resp.StatusCode, attempt)
		} else {
			log.Printf("[WARN] Webhook %s delivery network error: %v on attempt %d", event.ID, err, attempt)
		}

		// Exponential backoff with jitter
		time.Sleep(backoff)
		backoff *= 2
	}

	return errors.New("max retry attempts exceeded for webhook delivery")
}

// BatchProcessor handles concurrent dispatching of multiple webhook events.
func (d *Dispatcher) BatchProcessor(ctx context.Context, targetURL string, events []WebhookEvent) {
	var wg sync.WaitGroup
	semaphore := make(chan struct{}, 10) // Limit concurrency to 10 parallel workers

	for _, ev := range events {
		wg.Add(1)
		go func(event WebhookEvent) {
			defer wg.Done()
			semaphore <- struct{3}{}
			defer func() { <-semaphore }()

			if err := d.SendWithRetry(ctx, targetURL, event, 3); err != nil {
				log.Printf("[ERROR] Permanent failure for event %s: %v", event.ID, err)
			}
		}(ev)
	}
	wg.Wait()
}

func main() {
	dispatcher := NewDispatcher("super-secret-hmac-key", 5*time.Second)
	ctx := context.Background()

	events := []WebhookEvent{
		{ID: "evt_001", EventType: "mandate.signed", Payload: `{"mandate_id":"M123"}`, Timestamp: time.Now()},
		{ID: "evt_002", EventType: "payment.collected", Payload: `{"payment_id":"P456"}`, Timestamp: time.Now()},
	}

	dispatcher.BatchProcessor(ctx, "https://httpbin.org/post", events)
}
```
