## 3. Production-Ready Code / Architectural Proof
Below is a production-grade Python script utilizing the `requests` library to audit and clean up stale Docker and npm artifacts within an Artifactory instance based on last-downloaded timestamps and retention policies, embodying enterprise storage hygiene.

```python
#!/usr/bin/env python3
"""
Artifactory Storage Optimization & Artifact Retention Auditor
Author: Principal Systems Architect
Description: Scans JFrog Artifactory repositories for stale artifacts exceeding 
             retention thresholds and safely purges them via the Artifactory REST API.
"""

import logging
import sys
from typing import Any, Dict, List
import requests

# Configure structured logging
logging.basicConfig(
    stream=sys.stdout,
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(name)s: %(message)s",
)
logger = logging.getLogger("ArtifactoryAuditor")


class ArtifactoryAuditor:
    def __init__(self, base_url: str, api_token: str, dry_run: bool = True):
        self.base_url = base_url.rstrip("/")
        self.dry_run = dry_run
        self.session = requests.Session()
        self.session.headers.update(
            {
                "Authorization": f"Bearer {api_token}",
                "Content-Type": "application/json",
            }
        )

    def execute_AQL_query(self, aql_query: str) -> List[Dict[str, Any]]:
        """Executes an Artifactory Query Language (AQL) statement."""
        endpoint = f"{self.base_url}/api/search/aql"
        try:
            response = self.session.post(endpoint, data=aql_query, timeout=30)
            response.raise_for_status()
            return response.json().get("results", [])
        except requests.exceptions.RequestException as e:
            logger.error(f"AQL query execution failed: {e}")
            raise

    def find_stale_artifacts(self, repo_name: str, days_unused: int = 90) -> List[Dict[str, Any]]:
        """Finds artifacts not downloaded in the specified number of days."""
        aql_query = f"""
        items.find({{
            "repo": "{repo_name}",
            "type": "file",
            "stat.downloaded": {{"$before": "{days_unused}d"}}
        }}).include("repo", "path", "name", "stat.downloaded", "size")
        """
        logger.info(f"Querying repository '{repo_name}' for artifacts untouched for {days_unused} days...")
        return self.execute_AQL_query(aql_query)

    def purge_artifact(self, repo: str, path: str, name: str) -> bool:
        """Deletes an artifact from Artifactory."""
        file_path = f"{path}/{name}" if path != "." else name
        uri = f"{self.base_url}/{repo}/{file_path}"

        if self.dry_run:
            logger.info(f"[DRY-RUN] Would delete artifact: {uri}")
            return True

        try:
            response = self.session.delete(uri, timeout=15)
            if response.status_code == 204:
                logger.info(f"Successfully deleted: {uri}")
                return True
            else:
                logger.warning(f"Failed to delete {uri}. Status code: {response.status_code}")
                return False
        except requests.exceptions.RequestException as e:
            logger.error(f"Network error while deleting {uri}: {e}")
            return False


if __name__ == "__main__":
    # Configuration Example
    ARTIFACTORY_URL = "https://jfrog.internal.enterprise/artifactory"
    API_TOKEN = "ey123456789EXAMPLE_TOKEN"
    TARGET_REPO = "docker-local-repo"
    RETENTION_DAYS = 120

    auditor = ArtifactoryAuditor(base_url=ARTIFACTORY_URL, api_token=API_TOKEN, dry_run=True)
    
    try:
        stale_items = auditor.find_stale_artifacts(repo_name=TARGET_REPO, days_unused=RETENTION_DAYS)
        logger.info(f"Found {len(stale_items)} stale artifacts matching criteria.")
        
        for item in stale_items:
            auditor.purge_artifact(repo=item["repo"], path=item["path"], name=item["name"])
            
    except Exception as exc:
        logger.fatal(f"Audit execution aborted due to unhandled exception: {exc}")
        sys.exit(1)
```
