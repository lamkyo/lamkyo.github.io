## 3. Production-Ready Code / Architectural Proof

Below is a production-grade, highly concurrent Go implementation demonstrating a core component of Samsara’s telematics processing pipeline: an asynchronous, batching event processor with backpressure management, structured context propagation, and metric emission.

```go
// Package telematics implements a high-throughput, concurrent event ingestion and 
// batching processor designed for real-time IoT and fleet sensor streams.
package telematics

import (
	"context"
	"errors"
	"fmt"
	"sync"
	"sync/atomic"
	"time"

	"go.uber.org/zap"
)

// TelemetryEvent represents a normalized data packet from an edge device or sensor.
type TelemetryEvent struct {
	DeviceID  string                 `json:"device_id"`
	Timestamp time.Time              `json:"timestamp"`
	Metric    string                 `json:"metric"`
	Value     float64                `json:"value"`
	Metadata  map[string]interface{} `json:"metadata,omitempty"`
}

// BatchProcessorConfig holds tuning parameters for batch flushing.
type BatchProcessorConfig struct {
	BatchSize     int
	FlushInterval time.Duration
	WorkerPoolSize int
	QueueCapacity int
}

// Sink defines the interface for persisting processed event batches.
type Sink interface {
	WriteBatch(ctx context.Context, batch []TelemetryEvent) error
}

// Processor manages concurrent ingestion, batching, and sink persistence.
type Processor struct {
	config  BatchProcessorConfig
	sink    Sink
	logger  *zap.Logger
	inputCh chan TelemetryEvent
	wg      sync.WaitGroup
	dropped int64
	closed  uint32
}

// NewProcessor initializes a new Telemetry Processor instance.
func NewProcessor(config BatchProcessorConfig, sink Sink, logger *zap.Logger) *Processor {
	return &Processor{
		config:  config,
		sink:    sink,
		logger:  logger,
		inputCh: make(chan TelemetryEvent, config.QueueCapacity),
	}
}

// Start spawns worker routines responsible for batching and flushing events.
func (p *Processor) Start(ctx context.Context) {
	for i := 0; i < p.config.WorkerPoolSize; i++ {
		p.wg.Add(1)
		go p.worker(ctx, i)
	}
	p.logger.Info("Telemetry processor started", zap.Int("workers", p.config.WorkerPoolSize))
}

// Submit ingests a new event into the processing pipeline with non-blocking backpressure check.
func (p *Processor) Submit(event TelemetryEvent) error {
	if atomic.LoadUint32(&p.closed) == 1 {
		return errors.New("processor is shut down")
	}

	select {
	case p.inputCh <- event:
		return nil
	default:
		// Backpressure hit: increment dropped counter and return error or apply drop policy
		atomic.AddInt64(&p.dropped, 1)
		p.logger.Warn("Telemetry queue capacity reached, event dropped", 
			zap.String("device_id", event.DeviceID),
			zap.Int64("total_dropped", atomic.LoadInt64(&p.dropped)),
		)
		return errors.New("ingestion queue saturated; backpressure applied")
	}
}

// worker aggregates events into batches and flushes them to the sink.
func (p *Processor) worker(ctx context.Context, workerID int) {
	defer p.wg.Done()

	batch := make([]TelemetryEvent, 0, p.config.BatchSize)
	ticker := time.NewTicker(p.config.FlushInterval)
	defer ticker.Stop()

	flush := func() {
		if len(batch) == 0 {
			return
		}
		flushCtx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
		defer cancel()

		if err := p.sink.WriteBatch(flushCtx, batch); err != nil {
			p.logger.Error("Failed to write telemetry batch to sink", 
				zap.Int("worker_id", workerID), 
				zap.Int("batch_size", len(batch)), 
				zap.Error(err),
			)
		}
		// Reset batch slice preserving allocated backing array
		batch = batch[:0]
	}

	for {
		select {
		case <-ctx.Done():
			// Drain remaining items on shutdown
			for {
				select {
				case ev := <-p.inputCh:
					batch = append(batch, ev)
					if len(batch) >= p.config.BatchSize {
						flush()
					}
				default:
					flush()
					return
				}
			}

		case ev := <-p.inputCh:
			batch = append(batch, ev)
			if len(batch) >= p.config.BatchSize {
				flush()
			}

		case <-ticker.C:
			flush()
		}
	}
}

// Stop gracefully shuts down the processor.
func (p *Processor) Stop() {
	atomic.StoreUint32(&p.closed, 1)
	close(p.inputCh)
	p.wg.Wait()
	p.logger.Info("Telemetry processor stopped cleanly")
}
```
