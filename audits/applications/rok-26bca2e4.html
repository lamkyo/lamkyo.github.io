"""
Empire Assets - Real-Time Crypto Volatility & Spread Monitor
Author: Principal Systems Architect
Description: High-performance async service for real-time market data analysis,
             calculating rolling volatility and anomaly Z-scores for trading desks.
"""

import asyncio
import logging
import json
import numpy as np
import pandas as pd
from typing import Dict, Any, List
from collections import deque

# Configure structured logging
logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(name)s: %(message)s"
)
logger = logging.getLogger("EmpireAssets.AnalyticsEngine")

class MarketAnalyticsEngine:
    def __init__(self, symbol: str, window_size: int = 50, z_threshold: float = 2.5):
        self.symbol = symbol
        self.window_size = window_size
        self.z_threshold = z_threshold
        self.price_buffer: deque = deque(maxlen=window_size)
        self.lock = asyncio.Lock()

    async def process_tick(self, tick_data: Dict[str, Any]) -> None:
        """
        Process incoming market ticks, update sliding window, and compute statistical anomalies.
        """
        async with self.lock:
            try:
                price = float(tick_data["price"])
                volume = float(tick_data["volume"])
                timestamp = tick_data["timestamp"]

                self.price_buffer.append(price)

                if len(self.price_buffer) < self.window_size:
                    logger.debug(f"Accumulating buffer: {len(self.price_buffer)}/{self.window_size}")
                    return

                # Convert buffer to pandas series for vectorised statistical operations
                series = pd.Series(self.price_buffer)
                mean = series.mean()
                std = series.std()

                if std == 0 or np.isnan(std):
                    return

                # Calculate Z-Score for the latest price point
                current_z_score = (price - mean) / std

                logger.info(
                    f"[{self.symbol}] Price: {price:.2f} | Mean: {mean:.2f} | "
                    f"Std: {std:.4f} | Z-Score: {current_z_score:.2f}"
                )

                if abs(current_z_score) >= self.z_threshold:
                    await self._trigger_anomaly_alert(timestamp, price, current_z_score)

            except KeyError as e:
                logger.error(f"Malformed tick payload missing key: {e}")
            except Exception as e:
                logger.exception(f"Unexpected error processing market tick: {e}")

    async def _trigger_anomaly_alert(self, timestamp: int, price: float, z_score: float) -> None:
        """
        Simulate webhook or internal alerting dispatch for trading desk mentors.
        """
        alert_payload = {
            "symbol": self.symbol,
            "timestamp": timestamp,
            "price": price,
            "z_score": round(z_score, 4),
            "severity": "HIGH" if abs(z_score) > 3.5 else "MEDIUM",
            "message": f"Statistical price anomaly detected on {self.symbol} with Z-Score {z_score:.2f}"

            "volume": round(random.uniform(0.1, 5.0), 4),

        await engine.process_tick(tick)

    await mock_exchange_feed(engine)

        asyncio.run(main())
